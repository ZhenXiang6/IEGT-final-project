\section{研究背景與動機}
% =====================================================================
\begin{frame}{研究背景與動機}
  市場上的影音串流(如 Netflix、Disney+),收費模式有兩種主要型態:
  \begin{itemize}
    \item \hl{單片租借制}(TVOD, Transactional Video on Demand)
    \item \hl{訂閱制}(SVOD, Subscription Video on Demand)
  \end{itemize}
  \vskip0.5em
  而在\textbf{客群策略}的維度可分為:
  \begin{itemize}
    \item \hl{新顧客獲取}(Acquisition):廣告投放、首月免費／優惠
    \item \hl{舊顧客留存}(Retention):演算法推薦優化、家庭方案、Bundle
  \end{itemize}
  \vskip0.6em
  \begin{block}{研究動機}
    這兩個維度都\textbf{與當前市佔率(Market Share)$s$ 密切相關}。本研究探討:廠商應如何\textbf{隨 $s$ 動態地}選擇收費模式與客群策略?
  \end{block}
\end{frame}

\begin{frame}{研究背景與動機－市場現有策略分析}
  \vskip0.8em
  \centering
  {\small\renewcommand{\arraystretch}{1.45}
  \begin{tabular}{l|l|l|l}
    \textbf{Platform} & \textbf{Market Share} & \textbf{Pricing} & \textbf{Customer strategy}\\\hline
    Netflix        & High          & Subscription  & Retention\\
    Disney+        & High to medium& Subscription  & Acquisition $\to$ Retention\\
    Apple TV+      & Low to medium & Subscription  & Acquisition\\
    YouTube Movies & Low           & Transactional & Acquisition\\
  \end{tabular}}
  \vskip1.0em
  \begin{flushleft}\footnotesize
  \textbf{觀察:}市佔率越高的平台越偏向\hl{訂閱制 + 留存};市佔率低的新進者偏向\hl{租借制／獲取}。
  Disney+ 隨市佔成長,策略由「獲取」轉向「留存」—  這正是本專案想要探討的\textbf{策略隨 $s$ 演化}。
  \end{flushleft}
\end{frame}

\begin{frame}{研究問題}
  影音串流平台兩個維度的核心策略,都與當前市佔率 $s$ (Market Share)密切相關:
  \begin{columns}[t]
    \column{0.5\textwidth}\begin{block}{收費模式 $P$}\footnotesize
      \textbf{S} = 訂閱制 SVOD(Netflix 月費)\\
      \textbf{T} = 單部購買 TVOD(YouTube Movies 單片租購)\end{block}
    \column{0.5\textwidth}\begin{block}{客群策略 $C$}\footnotesize
      \textbf{A} = 新客獲取(廣告、試用)\\
      \textbf{R} = 舊客留存(推薦、折扣、內容)\end{block}
  \end{columns}
  \vskip0.2em
  \begin{center}\small
  \begin{tabular}{c|cc}
     & $C=A$(獲取) & $C=R$(留存)\\\hline
    $P=T$(TVOD) & \chip{ottred}\ \textcolor{ottred}{\textbf{Trans-Acq}} & \chip{ottorange}\ \textcolor{ottorange}{\textbf{Trans-Ret}}\\
    $P=S$(SVOD) & \chip{ottteal}\ \textcolor{ottteal}{\textbf{Sub-Acq}} & \chip{ottdark}\ \textcolor{ottdark}{\textbf{Sub-Ret}}\\
  \end{tabular}\end{center}
  \begin{alertblock}{核心問題}
    給定 $s$,廠商應如何在 4 種組合 $(P,C)\in\{T,S\}\times\{A,R\}$ 中選擇,以最大化跨期折現利潤?最適策略如何\textbf{隨 $s$ 變化}?
  \end{alertblock}
\end{frame}

\section{文獻回顧}
% =====================================================================
\begin{frame}{文獻回顧－ Two Dimension}
  \footnotesize\vskip0.1em
  \textbf{\color{rblue}Dimension 1:收費模式選擇(SVOD vs TVOD)}\par\smallskip
  \refx{Baek, Jang \& Kim (2024)}{Digital Business}{影片由 TVOD 移入 SVOD,觀看 $+192\%$、TVOD 銷售 $-52\%$(SVOD/TVOD 替代)。}
  \refx{Balasubramanian, Bhattacharya \& Krishnan (2015)}{Marketing Science}{selling(類 SVOD)vs pay-per-use(類 TVOD):獨佔利後者、競爭利前者。}
  \refx{Rao (2015)}{Marketing Science}{無承諾下「買斷(類 TVOD)」與「租用(類 SVOD)」並存最適,間接價格歧視。}
  \smallskip
  \textbf{\color{rblue}Dimension 2:客群策略(Acqusision vs Retention)}\par\smallskip
  \refx{Ben Rhouma \& Zaccour (2018)}{Management Science}{動態規劃最佳化獲取與留存支出(電信 panel 實證)。}
  \refx{Reinartz, Thomas \& Kumar (2005)}{J.\ of Marketing}{獲取/留存資源最適配置隨顧客關係階段而異。}
  \refx{Datta, Foubert \& Van Heerde (2015)}{JMR}{Free-trail客戶 CLV 低 59\%、存續期約三分之一。}
\end{frame}

\begin{frame}{文獻回顧－模型模板}
  \scriptsize\setlength{\smallskipamount}{1pt}\vskip0.1em
  \textbf{\color{rblue}\footnotesize 方法論模板:市佔率為狀態 + Bellman/DP 求解}\par\vskip1pt
  \refx{Sethi (1983)}{Optimal Control Appl.\ \& Methods}{市佔率 $x\in[0,1]$ 為狀態、廣告為控制,求 optimal $u^*(x)$}
  \refx{Ovchinnikov, Boulu-Reshef \& Pfeifer (2014)}{Management Science}{顧客基數為狀態、離散時間 Bellman 動態規劃,最適獲取/留存支出隨狀態調整。}
  \refx{Erickson (1992)}{Management Science}{Lanchester 市佔率動態 $\dot x{=}k_1u_1(1{-}x){-}k_2u_2x$(賽局模板)。}
  \vskip2pt
  \textbf{\color{rblue}\footnotesize 模型機制(網路外部性／規模成本)}\par\vskip1pt
  \refxn{Katz \& Shapiro (1985)／Klemperer (1987)／Farrell \& Klemperer (2007)}{網路外部性＋轉換成本}
  \refxn{Min et al.\ (2016)／Blattberg \& Deighton (1996)}{獲取成本隨市佔率遞增、留存並無此效應。}
\end{frame}

\section{模型建構}
% =====================================================================
\begin{frame}{模型建構－環境設定}
  \begin{columns}[t]
    \column{0.56\textwidth}\footnotesize
      \begin{itemize}
        \item 單一廠商;離散時間 $t=0,1,2,\dots$(一期為一策略週期)
        \item 折現利率$\beta\in(0,1)$,取 $\beta=0.95$
        \item 潛在市場標準化 $N\equiv1$;市佔率 $s_t\in[0,1]$
        \item 可被獲取的潛在客戶: $1-s_t$
      \end{itemize}
      \vskip0.4em
      每期同時選兩個維度
      \[ a_t=(P_t,C_t),\quad P_t\in\{\mathrm T,\mathrm S\},\ C_t\in\{\mathrm A,\mathrm R\} \]
      \[ \mathcal A=\{(\mathrm{T,A}),(\mathrm{T,R}),(\mathrm{S,A}),(\mathrm{S,R})\} \]
    \column{0.44\textwidth}
      \begin{block}{$2\times2$ action matrix}\centering\footnotesize
        \begin{tabular}{c|cc}
          & $C{=}A$ & $C{=}R$\\\hline
          $P{=}T$ & \chip{ottred} TA & \chip{ottorange} TR\\
          $P{=}S$ & \chip{ottteal} SA & \chip{ottdark} SR\\
        \end{tabular}\end{block}
  \end{columns}
\end{frame}

\begin{frame}{模型建構－狀態轉移(市佔率)}
  \[ \boxed{\,s_{t+1}=\underbrace{\rho(a_t,s_t)}_{\text{留存率}}\cdot s_t \;+\; \underbrace{\alpha(a_t,s_t)}_{\text{獲取率}}\cdot(1-s_t)\,} \]
  \begin{itemize}
    \item 第一項 $\rho\,s$:本期留下的\textbf{既有客戶}(留存率 $\rho$)。
    \item 第二項 $\alpha(1-s)$:從\textbf{非用戶池} $1-s$ 獲取的新客(獲取率 $\alpha$)。
    \item 以上兩項 ($\rho, \alpha$) 皆會根據 action ($a$) 與 market share ($s$) 而有變動。
  \end{itemize}
  \vskip0.3em
  {\footnotesize 下面兩頁分別給出 $\rho$、$\alpha$ 與利潤的具體函數形式}
\end{frame}

\begin{frame}{模型建構－顧客留存率(Retention Rate $\rho$)}
  \[ \boxed{\,\rho(P,C,s)=\underbrace{\rho_0}_{\text{base rate}}+\underbrace{\delta_P+\delta_C}_{\text{strategy diff}}+\mhl{\kappa_\rho\, s}} \]
  \begin{columns}[T]
    \column{0.5\textwidth}\centering\small
    \begin{tabular}{c|cc}
       & $C{=}R$ & $C{=}A$\\\hline
      $P{=}S$ & $\rho_0{+}\delta_S{+}\delta_R{+}\kappa_\rho s$ & $\rho_0{+}\delta_S{+}\kappa_\rho s$\\
      $P{=}T$ & $\rho_0{+}\delta_R{+}\kappa_\rho s$ & $\rho_0{+}\kappa_\rho s$\\
    \end{tabular}\\[0.4em]
    {\scriptsize $\rho_0{=}0.65$,$\delta_S{=}{+}0.105$(Iyengar 2011),$\delta_R{=}{+}0.10$,$\kappa_\rho{=}0.08$}
    \column{0.5\textwidth}
    \begin{block}{$\kappa_\rho\,s$ (network externality)}\footnotesize
      平台越大 $\Rightarrow$ 內容庫、推薦、社群效用越高 $\Rightarrow$ 留客率上升。\\
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{模型建構－顧客獲取率(Acquisition rate $\alpha$)}
  \[ \boxed{\,\alpha(P,C,s)=\underbrace{\alpha_0}_{\text{base rate}}+\underbrace{\tilde\delta_P+\tilde\delta_C}_{\text{strategy diff}}+\mhl{\kappa_\alpha\, s}} \]
  \begin{columns}[T]
    \column{0.52\textwidth}\centering\small
    \begin{tabular}{c|cc}
       & $C{=}R$ & $C{=}A$\\\hline
      $P{=}S$ & $\alpha_0{+}\tilde\delta_S{+}\kappa_\alpha s$ & $\alpha_0{+}\tilde\delta_S{+}\tilde\delta_A{+}\kappa_\alpha s$\\
      $P{=}T$ & $\alpha_0{+}\tilde\delta_T{+}\kappa_\alpha s$ & $\alpha_0{+}\tilde\delta_T{+}\tilde\delta_A{+}\kappa_\alpha s$\\
    \end{tabular}\\[0.4em]
    {\scriptsize $\alpha_0{=}0.10$,$\tilde\delta_T{=}{+}0.05$,$\tilde\delta_A{=}{+}0.15$,$\kappa_\alpha{=}0.20$}
    \column{0.48\textwidth}
  \end{columns}
\end{frame}

\begin{frame}{模型建構－利潤函數}
  \[ \boxed{\,\pi(s,a)=\underbrace{R(s,P)}_{\text{每用戶收入}}\cdot\, s \;-\; \underbrace{C(C,s)}_{\text{行銷成本}}\,} \]
  \begin{itemize}
    \item $R(s,P)$ 為每用戶每期收入,\quad $R(s,P)\cdot s$ 為總收入
    \item $C(C,s)$ 為每期行銷成本,\textbf{隨市佔率 $s$ 變動}。
  \end{itemize}
  \vskip0.4em
  \begin{block}{trade-off}\footnotesize
    \textbf{短期 vs.\ 跨期}:同 pricing 下留存(R)短期成本低;但獲取(A)提高 $\alpha$、推升未來 $s$,有跨期利益。\\
  \end{block}
\end{frame}

\begin{frame}{模型建構－收入函數}
  \[ \boxed{\,R(s,\mathrm S)=\underbrace{p_S}_{\text{規模無感}},\qquad R(s,\mathrm T)=\underbrace{p_T\,(\varphi_0+\varphi_1\, s)}_{\text{規模敏感}},\quad \varphi_1>0\,} \]
  \begin{columns}[T]
    \column{0.5\textwidth}\footnotesize
    \textbf{SVOD($p_S$):規模無感的均一收費。}\\
    消費者存在 flat-rate bias,願為固定月費\textbf{溢價}(Lambrecht \& Skiera 2006);大型捆綁使 $p_S$ 具競爭韌性(Bakos \& Brynjolfsson 1999/2000)。
    \column{0.5\textwidth}\footnotesize
    \textbf{TVOD($p_T(\varphi_0{+}\varphi_1 s)$):規模敏感。}\\
    平台越大 $\Rightarrow$ 目錄／受眾越大 $\Rightarrow$ 人均購買頻率上升;Carroni \& Paolini (2020) 證受眾達臨界值後平台轉向純訂閱($T\!\to\!S$)。
  \end{columns}
  \vskip0.5em
  \begin{block}{校準與依據}\footnotesize
    $p_S{=}15$、$p_T{=}5$、$\varphi_0{=}\varphi_1{=}0.5$;低 $s$ 時 $R(s,S)>R(s,T)$。Iyengar et al.\ (2011) 實證 two-part tariff 雖使留存 $-10.5\%$、用量 $-38.7\%$,但固定費收益補償使\textbf{整體利潤仍較高} —— 支撐 $p_S$ 高於邊際成本。
  \end{block}
\end{frame}

\begin{frame}{模型建構－成本函數(市場規模)}
  \[ \boxed{\,C_A(s)=c_A\,(1+\lambda_A\, s),\qquad C_R(s)=c_R\,(1+\lambda_R\, s),\qquad \lambda_A>\lambda_R\,} \]
  \begin{columns}[T]
    \column{0.52\textwidth}
    \begin{block}{為何 $\lambda_A>\lambda_R$}\footnotesize
      Min et al.\ (2016) 以 41 國無線電信實證:市場領導者的\textbf{獲取成本優勢隨滲透率上升}(高 $s$ 時拉新更貴),但\textbf{留存成本無此規模效應}。\\
      $\Rightarrow$ 市場越飽和,廠商越\textbf{提早轉向留存}。
    \end{block}
    \column{0.48\textwidth}\centering
    \includegraphics[width=0.70\linewidth]{fig_cost.png}\\
    {\scriptsize $C_A(s)$(紅)陡升、$C_R(s)$(橘)緩升。}
  \end{columns}
  \vskip0.2em
  {\footnotesize 校準:$c_A{=}1.2,\lambda_A{=}1.0$($C_A$:$1.2\!\to\!2.4$);$c_R{=}0.8,\lambda_R{=}0.3$($C_R$:$0.8\!\to\!1.04$)。}
\end{frame}

\begin{frame}{模型建構－Bellman equation與求解方法}
  \[ \boxed{\,V(s)=\max_{a\in\mathcal A}\big\{\,\pi(s,a)+\beta\,V(s'(s,a))\,\big\}\,},\quad s'=\rho(a,s)\,s+\alpha(a,s)(1-s) \]
  \begin{columns}[T]
    \column{0.52\textwidth}
    \begin{block}{存在性與唯一性}\footnotesize
      $\mathcal A$ 有限、$[0,1]$ 緊緻、$\pi$ 連續有界、$\beta\in(0,1)$ $\Rightarrow$ Bellman 算子為\textbf{壓縮映射},唯一不動點 $V^*$ 存在(Bellman 1957;Blackwell 1965;SLP 1989)。
    \end{block}
    \column{0.48\textwidth}
    \begin{exampleblock}{Value Iteration}\footnotesize
      1. 於 $s\in\{0,0.0025,\dots,1\}$(\textbf{401 格點})初始化 $V_0{=}0$。\\
      2. 每格 $\times$ 每 $a$:$Q=\pi+\beta\,V_k(s')$,$V_{k+1}=\max_a Q$,以線性插值取 $V_k(s')$。\\
      3. $\lVert V_{k+1}-V_k\rVert_\infty<10^{-8}$ 停止。
    \end{exampleblock}
  \end{columns}
  \vskip0.2em
  {\footnotesize 本模型於 \hl{406 次迭代}收斂。VI 在整個 $[0,1]$ 同步求出最適策略函數 $a^*(\cdot)$,不需指定起點(起點僅用於事後模擬路徑)。}
\end{frame}

  最適策略 $a^*(s)$ 在強化模型下仍呈\textbf{乾淨的三段結構}:

\section{延伸模型與結果}
% =====================================================================
\begin{frame}{延伸模型與結果－網路效應與市場集中}
  \begin{columns}[T]
    \column{0.54\textwidth}\centering
      \includegraphics[width=\linewidth]{fig_bifurcation.png}
    \column{0.46\textwidth}\footnotesize
      掃描留存網路外部性 $\kappa_\rho$:穩態 $s_\infty$ 由 $\kappa_\rho{=}0$ 的 $\sim$0.60 一路升到 $\kappa_\rho{\ge}0.14$ 的 $\sim$0.94。
      \begin{alertblock}{Critical Mass / 集中化}
        留存網路效應越強,市場越\textbf{傾向 winner-take-most}:跨過臨界區後廠商幾乎獨佔。呼應 Rohlfs (1974) 臨界質量、Arthur (1989) lock-in。
      \end{alertblock}
      對應現實:串流市場在內容與推薦的網路效應下,長期趨向少數平台高度集中。
  \end{columns}
\end{frame}

\section{結論}
% =====================================================================
\begin{frame}{結論}
  \footnotesize
  \begin{enumerate}\itemsep3pt
    \item \textbf{三段策略地圖}:Trans-Acq $\to$ Sub-Acq $\to$ Sub-Ret,內生重現 OTT「試水 $\to$ 規模化拉新 $\to$ 成熟留客」生命週期,且在加入網路外部性與規模成本後\textbf{結構穩健}。
    \item \textbf{網路外部性推升集中}:留存 $\kappa_\rho s$ 與獲取 $\kappa_\alpha s$ 使穩態由 0.50 升至 0.74,$\kappa_\rho$ 夠強時趨向 winner-take-most(臨界質量)。
    \item \textbf{Bass 口碑的雙面性}:$\kappa_\alpha$ 對留存有利既有者、對獲取卻\textbf{幫助挑戰者追趕},incumbent advantage 因而被部分抵銷。
    \item \textbf{Dimension separation}:成本／網路效應主要移動\textbf{客群維度}($A/R$),收費維度($T/S$)由收入結構主導。
  \end{enumerate}
  \vskip0.3em
  \begin{exampleblock}{方法論貢獻}
    在同一 Bellman 框架下整合\hl{收費模式選擇}與\hl{客群策略},並讓收費模式 $P$ 同時影響 $\rho$ 與 $\alpha$;所有函數形式皆有\textbf{經查證文獻}支撐(Bass 擴散 $\times$ 網路外部性 $\times$ 規模成本)。
  \end{exampleblock}
\end{frame}
